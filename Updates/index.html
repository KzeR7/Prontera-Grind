Drop-In Integration for Your HTML5 Game

Copy these files from the workspace (ro-map-engine.js, ro-map-data.json, ro-spritesheet.png, and ro-spritesheet.json) into your game folder and initialize the renderer on any <canvas>: 

<canvas id="gameCanvas" width="1024" height="640"></canvas>
<script src="ro-map-engine.js"></script>
<script>
async function initRagnarokMap() {
  const [mapData, spriteManifest] = await Promise.all([
    fetch('./ro-map-data.json').then(r => r.json()),
    fetch('./ro-spritesheet.json').then(r => r.json())
  ]);

  const spriteSheetImage = new Image();
  spriteSheetImage.src = './ro-spritesheet.png';
  await spriteSheetImage.decode();

  const canvas = document.getElementById('gameCanvas');
  const map = new RagnarokMapRenderer(canvas, {
    mapData,
    spriteSheetImage,
    spriteManifest,
    showWalkGrid: false
  });

  // Click-to-move character with built-in A* pathfinding on .gat cells
  canvas.addEventListener('click', (e) => {
    const rect = canvas.getBoundingClientRect();
    const cell = map.pickCellAtScreen(e.clientX - rect.left, e.clientY - rect.top);
    if (cell && cell.walkable) {
      map.moveCharacterTo(cell.x + 0.5, cell.y + 0.5);
    }
  });

  map.start();
}
initRagnarokMap();
</script>
